# Book Warehouse Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first Book Warehouse parity slice: hidden-source foundation, shared contracts, database schema, encrypted admin settings, typed upstream client, fixtures, and an admin-only connection UI.

**Architecture:** Treat Book Warehouse as an internal source provider, not a user-facing product area. The server gets an internal `warehouse` module with settings, secret protection, repository, and client services. The client only adds an admin integration tab labelled as a native catalog source; regular Library, Discover, Search, Requests, reader, and dashboard surfaces remain untouched in this slice.

**Tech Stack:** NestJS, Fastify, Drizzle ORM, PostgreSQL, Vitest, Vue 3, Pinia-free feature APIs, `@bookorbit/types`, native `fetch`, Node `crypto`.

---

## Guardrails

- Do not add a regular user-facing "Warehouse" page, badge, tab, or source label.
- Do not expose `api_key` query authentication. The upstream client must send only the `X-API-Key` header.
- Do not return stored API keys from backend APIs. Return `apiKeyConfigured` and a masked preview only.
- Do not wire synced catalog data into normal user-facing search, dashboard, request, reader, or device surfaces in this slice.
- Use Book Warehouse terminology only in internal code, admin-only integration settings, logs, and developer docs.

```ts
import { BadRequestException, Injectable } from "@nestjs/common";
import { ConfigService } from "@nestjs/config";
import { createCipheriv, createDecipheriv, randomBytes } from "crypto";

export interface EncryptedWarehouseSecret {
  ciphertext: string;
  nonce: string;
  tag: string;
}

const HEX_256_BIT_KEY = /^[a-fA-F0-9]{64}$/;

@Injectable()
export class WarehouseSecretService {
  constructor(private readonly config: ConfigService) {}

  encrypt(value: string): EncryptedWarehouseSecret {
    const key = this.key();
    const nonce = randomBytes(12);
    const cipher = createCipheriv("aes-256-gcm", key, nonce);
    const ciphertext = Buffer.concat([
      cipher.update(value, "utf8"),
      cipher.final(),
    ]);
    const tag = cipher.getAuthTag();

    return {
      ciphertext: ciphertext.toString("base64"),
      nonce: nonce.toString("base64"),
      tag: tag.toString("base64"),
    };
  }

  decrypt(secret: EncryptedWarehouseSecret): string {
    const key = this.key();
    const decipher = createDecipheriv(
      "aes-256-gcm",
      key,
      Buffer.from(secret.nonce, "base64"),
    );
    decipher.setAuthTag(Buffer.from(secret.tag, "base64"));
    return Buffer.concat([
      decipher.update(Buffer.from(secret.ciphertext, "base64")),
      decipher.final(),
    ]).toString("utf8");
  }

  mask(value: string | null | undefined): string | null {
    if (!value) return null;
    if (value.length < 10) return "***";
    return `${value.slice(0, 3)}...${value.slice(-3)}`;
  }

  private key(): Buffer {
    const raw = this.config.get<string>("app.warehouse.encryptionKey")?.trim();
    if (!raw) {
      throw new BadRequestException(
        "WAREHOUSE_ENCRYPTION_KEY is required before saving a catalog source API key",
      );
    }

    if (!HEX_256_BIT_KEY.test(raw)) {
      throw new BadRequestException(
        "WAREHOUSE_ENCRYPTION_KEY must be a 64-character hex key before saving a catalog source API key",
      );
    }

    return Buffer.from(raw, "hex");
  }
}
```

```ts
import { BadRequestException, Injectable } from "@nestjs/common";
import type {
  UpsertWarehouseAdminSettingsPayload,
  WarehouseAdminSettings,
  WarehouseConnectionTestResult,
} from "@bookorbit/types";
import { WarehouseClientService } from "./warehouse-client.service";
import { WarehouseRepository } from "./warehouse.repository";
import { WarehouseSecretService } from "./warehouse-secret.service";

const DEFAULT_BASE_URL = "https://bookwarehouse.zenterprise.org";
const DEFAULT_SYNC_CADENCE_MINUTES = 360;

@Injectable()
export class WarehouseSettingsService {
  constructor(
    private readonly repo: WarehouseRepository,
    private readonly secret: WarehouseSecretService,
    private readonly client: WarehouseClientService,
  ) {}

  async getAdminSettings(): Promise<WarehouseAdminSettings> {
    const row = await this.repo.findSettings();
    if (!row) {
      return {
        enabled: false,
        baseUrl: DEFAULT_BASE_URL,
        apiKeyConfigured: false,
        apiKeyPreview: null,
        syncCadenceMinutes: DEFAULT_SYNC_CADENCE_MINUTES,
        lastConnectionStatus: "untested",
        lastConnectionCheckedAt: null,
        lastConnectionError: null,
      };
    }

    const apiKey =
      row.apiKeyEncrypted && row.apiKeyNonce && row.apiKeyTag
        ? this.secret.decrypt({
            ciphertext: row.apiKeyEncrypted,
            nonce: row.apiKeyNonce,
            tag: row.apiKeyTag,
          })
        : null;

    return {
      enabled: row.enabled,
      baseUrl: row.baseUrl,
      apiKeyConfigured: Boolean(apiKey),
      apiKeyPreview: this.secret.mask(apiKey),
      syncCadenceMinutes: row.syncCadenceMinutes,
      lastConnectionStatus:
        row.lastConnectionStatus as WarehouseAdminSettings["lastConnectionStatus"],
      lastConnectionCheckedAt:
        row.lastConnectionCheckedAt?.toISOString() ?? null,
      lastConnectionError: row.lastConnectionError,
    };
  }

  async upsertAdminSettings(
    payload: UpsertWarehouseAdminSettingsPayload,
  ): Promise<WarehouseAdminSettings> {
    const existing = await this.repo.findSettings();
    const nextEnabled = payload.enabled ?? existing?.enabled ?? false;
    const nextBaseUrl = normalizeBaseUrl(
      payload.baseUrl ?? existing?.baseUrl ?? DEFAULT_BASE_URL,
    );
    const encrypted = payload.apiKey?.trim()
      ? this.secret.encrypt(payload.apiKey.trim())
      : existing?.apiKeyEncrypted && existing.apiKeyNonce && existing.apiKeyTag
        ? {
            ciphertext: existing.apiKeyEncrypted,
            nonce: existing.apiKeyNonce,
            tag: existing.apiKeyTag,
          }
        : null;

    if (nextEnabled && !encrypted) {
      throw new BadRequestException(
        "Catalog source API key is required before enabling sync",
      );
    }

    await this.repo.upsertSettings({
      enabled: nextEnabled,
      baseUrl: nextBaseUrl,
      apiKeyEncrypted: encrypted?.ciphertext ?? null,
      apiKeyNonce: encrypted?.nonce ?? null,
      apiKeyTag: encrypted?.tag ?? null,
      syncCadenceMinutes:
        payload.syncCadenceMinutes ??
        existing?.syncCadenceMinutes ??
        DEFAULT_SYNC_CADENCE_MINUTES,
      lastConnectionStatus: existing?.lastConnectionStatus ?? "untested",
      lastConnectionCheckedAt: existing?.lastConnectionCheckedAt ?? null,
      lastConnectionError: existing?.lastConnectionError ?? null,
    });

    return this.getAdminSettings();
  }

  async testConnection(): Promise<WarehouseConnectionTestResult> {
    const row = await this.repo.findSettings();
    const apiKey =
      row?.apiKeyEncrypted && row.apiKeyNonce && row.apiKeyTag
        ? this.secret.decrypt({
            ciphertext: row.apiKeyEncrypted,
            nonce: row.apiKeyNonce,
            tag: row.apiKeyTag,
          })
        : null;

    if (!row || !apiKey) {
      throw new BadRequestException(
        "Catalog source API key is required before testing the connection",
      );
    }

    const result = await this.client.testConnection(row.baseUrl, apiKey);
    await this.repo.updateConnectionStatus(
      result.ok ? "ok" : "error",
      new Date(result.checkedAt),
      result.ok ? null : result.message,
    );
    return result;
  }
}

function normalizeBaseUrl(value: string): string {
  return value.trim().replace(/\/+$/, "");
}
```

```ts
import { Injectable } from "@nestjs/common";
import type { WarehouseConnectionTestResult } from "@bookorbit/types";
import {
  WAREHOUSE_API_PREFIX,
  WAREHOUSE_MAX_PAGE_LIMIT,
  WAREHOUSE_USER_AGENT,
} from "./warehouse.constants";
import { WarehouseApiError } from "./warehouse.errors";

interface BaseRequest {
  baseUrl: string;
  apiKey: string;
}

interface PageRequest extends BaseRequest {
  page: number;
  limit: number;
}

@Injectable()
export class WarehouseClientService {
  async testConnection(
    baseUrl: string,
    apiKey: string,
  ): Promise<WarehouseConnectionTestResult> {
    const checkedAt = new Date().toISOString();
    const response = await fetch(`${normalizeBaseUrl(baseUrl)}/health`, {
      headers: this.headers(apiKey),
    });

    if (!response.ok) {
      return {
        ok: false,
        status: response.status,
        message: await this.errorMessage(response),
        checkedAt,
      };
    }

    return {
      ok: true,
      status: response.status,
      message: "Connected",
      checkedAt,
    };
  }

  listBooks(request: PageRequest) {
    return this.getJson(request, "/books", {
      page: String(request.page),
      limit: String(limit(request.limit)),
    });
  }

  listAudiobooks(request: PageRequest) {
    return this.getJson(request, "/audiobooks", {
      page: String(request.page),
      limit: String(limit(request.limit)),
    });
  }

  searchExternalBooks(request: BaseRequest & { q: string }) {
    return this.getJson(request, "/search/external", { q: request.q });
  }

  searchExternalAudiobooks(request: BaseRequest & { q: string }) {
    return this.getJson(request, "/audiobooks/search/external", {
      q: request.q,
    });
  }

  searchAbiplayerAudiobooks(request: BaseRequest & { q: string }) {
    return this.getJson(request, "/audiobooks/abiplayer/search", {
      q: request.q,
    });
  }

  requestBook(
    request: BaseRequest & {
      isbn?: string;
      preferred_format?: string;
      search_result?: Record<string, unknown>;
    },
  ) {
    return this.postJson(request, "/monitoring/add", {
      isbn: request.isbn,
      preferred_format: request.preferred_format,
      search_result: request.search_result,
    });
  }

  requestAudiobook(request: BaseRequest & { title: string; author?: string }) {
    return this.postJson(request, "/audiobooks/abiplayer/requests", {
      title: request.title,
      ...(request.author ? { author: request.author } : {}),
    });
  }

  private async getJson(
    request: BaseRequest,
    path: string,
    params: Record<string, string>,
  ) {
    const url = new URL(
      `${normalizeBaseUrl(request.baseUrl)}${WAREHOUSE_API_PREFIX}${path}`,
    );
    for (const [key, value] of Object.entries(params))
      url.searchParams.set(key, value);
    return this.fetchJson(url.toString(), request.apiKey);
  }

  private async postJson(
    request: BaseRequest,
    path: string,
    body: Record<string, unknown>,
  ) {
    return this.fetchJson(
      `${normalizeBaseUrl(request.baseUrl)}${WAREHOUSE_API_PREFIX}${path}`,
      request.apiKey,
      {
        method: "POST",
        body: JSON.stringify(body),
      },
    );
  }

  private async fetchJson(url: string, apiKey: string, init: RequestInit = {}) {
    const response = await fetch(url, {
      ...init,
      headers: {
        ...this.headers(apiKey),
        ...(init.body ? { "Content-Type": "application/json" } : {}),
      },
    });

    if (!response.ok)
      throw new WarehouseApiError(
        response.status,
        await this.errorMessage(response),
      );
    return response.json();
  }

  private headers(apiKey: string): Record<string, string> {
    return {
      "X-API-Key": apiKey,
      "User-Agent": WAREHOUSE_USER_AGENT,
    };
  }

  private async errorMessage(response: Response): Promise<string> {
    try {
      const json = (await response.json()) as { error?: unknown };
      return typeof json.error === "string" ? json.error : response.statusText;
    } catch {
      return response.statusText;
    }
  }
}

function normalizeBaseUrl(baseUrl: string): string {
  return baseUrl.trim().replace(/\/+$/, "");
}

function limit(value: number): number {
  return Math.max(1, Math.min(WAREHOUSE_MAX_PAGE_LIMIT, value));
}
```

```ts
import { Body, Controller, Get, Patch, Post } from "@nestjs/common";
import { Permission } from "@bookorbit/types";
import { RequirePermission } from "../../common/decorators/require-permission.decorator";
import { UpsertWarehouseAdminSettingsDto } from "./dto";
import { WarehouseSettingsService } from "./warehouse-settings.service";

@RequirePermission(Permission.ManageAppSettings)
@Controller("admin/warehouse")
export class WarehouseAdminController {
  constructor(private readonly settings: WarehouseSettingsService) {}

  @Get("settings")
  getSettings() {
    return this.settings.getAdminSettings();
  }

  @Patch("settings")
  updateSettings(@Body() dto: UpsertWarehouseAdminSettingsDto) {
    return this.settings.upsertAdminSettings(dto);
  }

  @Post("test-connection")
  testConnection() {
    return this.settings.testConnection();
  }
}
```

```ts
import { api } from "@/lib/api";
import type {
  UpsertWarehouseAdminSettingsPayload,
  WarehouseAdminSettings,
  WarehouseConnectionTestResult,
} from "@bookorbit/types";

async function expectJson<T>(
  response: Response,
  fallbackMessage: string,
): Promise<T> {
  if (response.ok) return response.json() as Promise<T>;
  const payload = await response.json().catch(() => ({}));
  const message =
    payload &&
    typeof payload === "object" &&
    "message" in payload &&
    typeof payload.message === "string"
      ? payload.message
      : fallbackMessage;
  throw new Error(message);
}

export async function fetchWarehouseAdminSettings(): Promise<WarehouseAdminSettings> {
  return expectJson<WarehouseAdminSettings>(
    await api("/api/v1/admin/warehouse/settings"),
    "Failed to load catalog source settings",
  );
}

export async function updateWarehouseAdminSettings(
  payload: UpsertWarehouseAdminSettingsPayload,
): Promise<WarehouseAdminSettings> {
  return expectJson<WarehouseAdminSettings>(
    await api("/api/v1/admin/warehouse/settings", {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    }),
    "Failed to save catalog source settings",
  );
}

export async function testWarehouseConnection(): Promise<WarehouseConnectionTestResult> {
  return expectJson<WarehouseConnectionTestResult>(
    await api("/api/v1/admin/warehouse/test-connection", { method: "POST" }),
    "Failed to test catalog source connection",
  );
}
```

```vue
<script setup lang="ts">
import { onMounted, reactive, ref } from "vue";
import { toast } from "vue-sonner";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import {
  fetchWarehouseAdminSettings,
  testWarehouseConnection,
  updateWarehouseAdminSettings,
} from "../api/warehouse-admin.api";

const loading = ref(true);
const saving = ref(false);
const testing = ref(false);
const apiKeyPlaceholder = ref("");

const form = reactive({
  enabled: false,
  baseUrl: "https://bookwarehouse.zenterprise.org",
  apiKey: "",
  syncCadenceMinutes: 360,
  apiKeyConfigured: false,
  apiKeyPreview: null as string | null,
  lastConnectionStatus: "untested",
  lastConnectionError: null as string | null,
});

async function load() {
  loading.value = true;
  const settings = await fetchWarehouseAdminSettings();
  form.enabled = settings.enabled;
  form.baseUrl = settings.baseUrl;
  form.syncCadenceMinutes = settings.syncCadenceMinutes;
  form.apiKeyConfigured = settings.apiKeyConfigured;
  form.apiKeyPreview = settings.apiKeyPreview;
  form.lastConnectionStatus = settings.lastConnectionStatus;
  form.lastConnectionError = settings.lastConnectionError;
  apiKeyPlaceholder.value = settings.apiKeyConfigured
    ? `Stored key ${settings.apiKeyPreview ?? ""}`
    : "Paste API key";
  loading.value = false;
}

async function save() {
  saving.value = true;
  try {
    const settings = await updateWarehouseAdminSettings({
      enabled: form.enabled,
      baseUrl: form.baseUrl,
      apiKey: form.apiKey.trim() || undefined,
      syncCadenceMinutes: form.syncCadenceMinutes,
    });
    form.apiKey = "";
    form.apiKeyConfigured = settings.apiKeyConfigured;
    form.apiKeyPreview = settings.apiKeyPreview;
    toast.success("Catalog source settings saved");
  } finally {
    saving.value = false;
  }
}

async function testConnection() {
  testing.value = true;
  try {
    const result = await testWarehouseConnection();
    toast[result.ok ? "success" : "error"](result.message);
  } finally {
    testing.value = false;
  }
}

onMounted(load);
</script>
```
